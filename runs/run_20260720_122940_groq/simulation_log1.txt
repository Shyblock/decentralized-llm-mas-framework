        for i in range(n + 1):
            pref = w[:i]
            prefix_dict.setdefault(pref, []).append(w)
    results = []

    def backtrack(step, path):
        if step == n:
            results.append(path[:])
            return
        prefix = ''.join((path[i][step] for i in range(step)))
        for cand in prefix_dict.get(prefix, []):
            path.append(cand)
            backtrack(step + 1, path)
            path.pop()
    for w in words:
        backtrack(1, [w])
    return results

            orig_parent = parent[x]

        if x not in parent:
            parent[x] = x
            weight[x] = 1.0
        if y not in parent:
            parent[y] = y
            weight[y] = 1.0

        ry, wy = find(y)
        if rx == ry:
            return
        parent[rx] = ry
        weight[rx] = value * wy / wx
    for (a, b), v in zip(equations, values):
        union(a, b, v)

    for a, b in queries:
        if a not in parent or b not in parent:

def solve(s):
    n = len(s)
    if n == 0:
        return 0
    char_to_idx = lambda ch: ord(ch) - 97
    next_idx = [[n] * 26 for _ in range(n + 1)]
    for i in range(n - 1, -1, -1):
        for c in range(26):
            next_idx[i][c] = next_idx[i + 1][c]
        next_idx[i][char_to_idx(s[i])] = i
    prev_idx = [[-1] * 26 for _ in range(n)]
    for i in range(n):
        if i > 0:
            for c in range(26):
                prev_idx[i][c] = prev_idx[i - 1][c]
        prev_idx[i][char_to_idx(s[i])] = i
    dp = [[0] * n for _ in range(n)]
    for i in range(n):
        dp[i][i] = 1
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j]:
                ch = char_to_idx(s[i])
                low = next_idx[i + 1][ch]
                high = prev_idx[j - 1][ch]
                if low > high:
                    dp[i][j] = dp[i + 1][j - 1] * 2 + 2
                elif low == high:
                    dp[i][j] = dp[i + 1][j - 1] * 2 + 1
                else:
                    dp[i][j] = dp[i + 1][j - 1] * 2 - dp[low + 1][high - 1]
            else:
                dp[i][j] = dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1]
            if dp[i][j] < 0:
                dp[i][j] = 0
    return dp[0][n - 1]

def solve(nums, k):
    total = sum(nums)
    if k == 0 or total % k != 0:
        return False

        return False

    if nums[0] > target:
        return False
    used = [False] * n

    def backtrack(start_index, cur_sum, groups_done):
        if groups_done == k - 1:
            return True
        if cur_sum == target:
            return backtrack(0, 0, groups_done + 1)
        prev = -1
        for i in range(start_index, n):
            if not used[i] and cur_sum + nums[i] <= target and (nums[i] != prev):
                used[i] = True
                if backtrack(i + 1, cur_sum + nums[i], groups_done):
                    return True
                used[i] = False
                prev = nums[i]
                if cur_sum == 0 or cur_sum + nums[i] == target:
                    break
        return False
    return backtrack(0, 0, 0)

        if target in routes[bus]:
            return cnt

                    visited_buses.add(nb)
                    q.append((nb, cnt + 1))
    return -1

def solve(equations, values, queries):
    graph = defaultdict(list)
    for (a, b), v in zip(equations, values):
        graph[a].append((b, v))
        graph[b].append((a, 1.0 / v))

    def bfs(src, dst):
        if src not in graph or dst not in graph:
            return -1.0
        if src == dst:
            return 1.0
        visited = {src}
        q = deque([(src, 1.0)])
        while q:
            node, prod = q.popleft()
            if node == dst:
                return prod
            for nei, w in graph[node]:
                if nei not in visited:
                    visited.add(nei)
                    q.append((nei, prod * w))
        return -1.0
    return [bfs(a, b) for a, b in queries]

def solve(arr):

        return 0

    i = 1
    while i < n - 1:
        if arr[i - 1] < arr[i] > arr[i + 1]:
            left = i - 1
            while left > 0 and arr[left - 1] < arr[left]:
                left -= 1
            right = i + 1
            while right < n - 1 and arr[right] > arr[right + 1]:
                right += 1
            cur_len = right - left + 1
            if cur_len > max_len:
                max_len = cur_len
