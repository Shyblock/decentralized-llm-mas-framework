            results.append(square[:])
            return
        prefix = ''.join((square[i][step] for i in range(step)))

def solve(equations, values, queries):
    graph = defaultdict(list)
    for (a, b), v in zip(equations, values):
        graph[a].append((b, v))
        graph[b].append((a, 1.0 / v))

    def bfs(src, dst):
        if src == dst:
            return 1.0
        visited = set([src])
        q = deque([(src, 1.0)])
        while q:
            node, prod = q.popleft()
            for nei, w in graph[node]:
                if nei in visited:
                    continue
                if nei == dst:
                    return prod * w
                visited.add(nei)
                q.append((nei, prod * w))
        return -1.0
    results = []
    for a, b in queries:
        if a not in graph or b not in graph:
            results.append(-1.0)
        elif a == b:
            results.append(1.0)
        else:
            results.append(bfs(a, b))
    return results

    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = (find(a), find(b))
        if ra == rb:
            return
        if size[ra] < size[rb]:
            ra, rb = (rb, ra)
        parent[rb] = ra
        size[ra] += size[rb]
    value_to_nodes = defaultdict(list)
    for i, v in enumerate(vals):
        value_to_nodes[v].append(i)
    unique_vals = sorted(value_to_nodes.keys())
    answer = n
    for val in unique_vals:
        nodes = value_to_nodes[val]
        for node in nodes:
            for nb in adj[node]:
                if vals[nb] <= val:
                    union(node, nb)
        comp_count = defaultdict(int)
        for node in nodes:
            root = find(node)
            comp_count[root] += 1
        for cnt in comp_count.values():
            answer += cnt * (cnt - 1) // 2
    return answer
