        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        ratio = weight[y] * value / weight[x]
        parent[root_x] = root_y
        weight[root_x] = ratio
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])

    result.sort(key=lambda g: g[0] if g else '')

def solve(num, target):
    results = []
    n = len(num)
    target = int(target)

    def dfs(index, path, eval_val, last):
        if index == n:
            if eval_val == target:
                results.append(path)
            return
        for i in range(index + 1, n + 1):
            cur_str = num[index:i]
            if len(cur_str) > 1 and cur_str[0] == '0':
                continue
            cur = int(cur_str)
            if index == 0:
                dfs(i, cur_str, cur, cur)
            else:
                dfs(i, path + '+' + cur_str, eval_val + cur, cur)
                dfs(i, path + '-' + cur_str, eval_val - cur, -cur)
                dfs(i, path + '*' + cur_str, eval_val - last + last * cur, last * cur)
    dfs(0, '', 0, 0)
    return results

def solve(s):
    MOD = 10 ** 9 + 7
    n = len(s)
    if n == 0:
        return 0
    next_pos = [[n] * 26 for _ in range(n)]
    prev_pos = [[-1] * 26 for _ in range(n)]
    last = [n] * 26
    for i in range(n - 1, -1, -1):
        last[ord(s[i]) - 97] = i
        for c in range(26):
            next_pos[i][c] = last[c]
    last = [-1] * 26
    for i in range(n):
        last[ord(s[i]) - 97] = i
        for c in range(26):
            prev_pos[i][c] = last[c]
    dp = [[0] * n for _ in range(n)]
    for i in range(n):
        dp[i][i] = 1
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] != s[j]:
                dp[i][j] = (dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1]) % MOD
            else:
                c = ord(s[i]) - 97
                ni = next_pos[i + 1][c] if i + 1 < n else n
                pj = prev_pos[j - 1][c] if j - 1 >= 0 else -1
                if ni > pj:
                    dp[i][j] = (dp[i + 1][j - 1] * 2 + 2) % MOD
                elif ni == pj:
                    dp[i][j] = (dp[i + 1][j - 1] * 2 + 1) % MOD
                else:
                    dp[i][j] = (dp[i + 1][j - 1] * 2 - dp[ni + 1][pj - 1]) % MOD
    return dp[0][n - 1] % MOD

                if loads[i] + cur <= limit and loads[i] not in seen:
                    seen.add(loads[i])

                        return True

            high = mid

            low = mid + 1

        if buckets_done == k - 1:
            return True
        if cur_sum == target:
            return backtrack(0, 0, buckets_done + 1)
        prev = -1
        i = start
        while i < n:
            if not used[i] and cur_sum + nums[i] <= target and (nums[i] != prev):
                used[i] = True
                if backtrack(i + 1, cur_sum + nums[i], buckets_done):
                    return True

                    return False
                if cur_sum + nums[i] == target:
                    return False
                prev = nums[i]
            i += 1
        return False

def solve(target):

    def dp(t):
        if t in memo:
            return memo[t]
        n = (t + 1).bit_length()
        if (1 << n) - 1 == t:
            memo[t] = n
            return n
        ans = n + 1 + dp((1 << n) - 1 - t)
        for m in range(n - 1):
            forward = (1 << n - 1) - 1
            back = (1 << m) - 1
            pos = forward - back
            if t > pos:
                remaining = t - pos
                ans = min(ans, n - 1 + 1 + m + 1 + dp(remaining))
        memo[t] = ans
        return ans
    return dp(target)

        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]

        rx, ry = (find(x), find(y))
        if rx == ry:
            return
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
    val_to_nodes = defaultdict(list)
    for i, v in enumerate(vals):
        val_to_nodes[v].append(i)
    answer = 0
    for value in sorted(val_to_nodes.keys()):

                if vals[nb] <= value:
                    union(node, nb)

        for node in nodes:
            root = find(node)
            comp_count[root] += 1
        for cnt in comp_count.values():
            answer += cnt * (cnt + 1) // 2

    for dest, src in prerequisites:
        if 0 <= src < num_courses and 0 <= dest < num_courses:
            graph[src].append(dest)
            indegree[dest] += 1
    q = deque([i for i in range(num_courses) if indegree[i] == 0])
    visited = 0
    while q:
        node = q.popleft()
        visited += 1
        for nei in graph[node]:

            if indegree[nei] == 0:
                q.append(nei)
    return visited == num_courses

        mask = 0
        for ch in set(w):
            mask |= 1 << ord(ch) - 97
        masks[i] = mask
        lengths[i] = len(w)
    max_product = 0
    for i in range(n):
        mi = masks[i]
        li = lengths[i]
        for j in range(i + 1, n):
            if mi & masks[j] == 0:
                prod = li * lengths[j]
                if prod > max_product:
                    max_product = prod

def solve(nums, target):

    def find_left(arr, t):
        lo, hi = (0, len(arr) - 1)
        ans = -1
        while lo <= hi:
            mid = (lo + hi) // 2
            if arr[mid] < t:
                lo = mid + 1
            else:
                if arr[mid] == t:
                    ans = mid
                hi = mid - 1
        return ans

    def find_right(arr, t):
        lo, hi = (0, len(arr) - 1)
        ans = -1
        while lo <= hi:
            mid = (lo + hi) // 2
            if arr[mid] > t:
                hi = mid - 1
            else:
                if arr[mid] == t:
                    ans = mid
                lo = mid + 1
        return ans
    left = find_left(nums, target)
    if left == -1:
        return [-1, -1]
    right = find_right(nums, target)
    return [left, right]

        visited.add((r, c))
        res = dfs(r + 1, c, idx + 1, visited) or dfs(r - 1, c, idx + 1, visited) or dfs(r, c + 1, idx + 1, visited) or dfs(r, c - 1, idx + 1, visited)
        visited.remove((r, c))
        return res
    for i in range(rows):
        for j in range(cols):
            if board[i][j] == word[0]:
                if dfs(i, j, 0, set()):
                    return True
    return False
