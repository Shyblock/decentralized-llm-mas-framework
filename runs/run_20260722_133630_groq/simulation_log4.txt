def solve(num_courses, prerequisites):
    adj = [[] for _ in range(num_courses)]
    indeg = [0] * num_courses
    for dest, src in prerequisites:
        if 0 <= src < num_courses and 0 <= dest < num_courses:
            adj[src].append(dest)
            indeg[dest] += 1
    q = deque([i for i in range(num_courses) if indeg[i] == 0])
    visited = 0
    while q:
        u = q.popleft()
        visited += 1
        for v in adj[u]:
            indeg[v] -= 1
            if indeg[v] == 0:
                q.append(v)
    return visited == num_courses

            if x >> b & 1:
                if cnt[b] == 0:
                    cur_or |= 1 << b
                cnt[b] += 1

            length = r - l + 1

                ans = length

                if y >> b & 1:
                    cnt[b] -= 1
                    if cnt[b] == 0:
                        cur_or &= ~(1 << b)
            l += 1
    return ans if ans != n + 1 else 0

def solve(graph):
    n = len(graph)
    if n == 0:
        return 0
    if n == 1:
        return 0
    all_mask = (1 << n) - 1
    visited = [[False] * (1 << n) for _ in range(n)]
    dq = deque()
    for i in range(n):
        mask = 1 << i
        visited[i][mask] = True
        dq.append((i, mask, 0))
    while dq:
        node, mask, dist = dq.popleft()
        if mask == all_mask:
            return dist
        for nb in graph[node]:
            new_mask = mask | 1 << nb
            if not visited[nb][new_mask]:
                visited[nb][new_mask] = True
                dq.append((nb, new_mask, dist + 1))
    return -1

    for start in range(1, n + 1):
        if color[start] != 0:
            continue
        color[start] = 1
        q = deque([start])
        while q:
            node = q.popleft()

def solve(nums):

    def merge_sort(arr):
        n = len(arr)
        if n <= 1:
            return arr[:]
        mid = n // 2
        left = merge_sort(arr[:mid])
        right = merge_sort(arr[mid:])
        return merge(left, right)

    def merge(left, right):
        i = j = 0
        merged = []
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i])
                i += 1
            else:
                merged.append(right[j])
                j += 1
        if i < len(left):
            merged.extend(left[i:])
        if j < len(right):
            merged.extend(right[j:])
        return merged
    return merge_sort(nums)

        return 1.0

    max_prob[start] = 1.0
    heap = [(-1.0, start)]
    while heap:
        cur_neg, node = heapq.heappop(heap)
        cur = -cur_neg
        if cur < max_prob[node]:
            continue
        if node == end:
            return cur
        for nei, p in adj[node]:
            new_prob = cur * p

                heapq.heappush(heap, (-new_prob, nei))
    return 0.0

def solve(vals, edges):
    n = len(vals)
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)
    parent = list(range(n))
    size = [1] * n

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = (find(a), find(b))
        if ra == rb:
            return
        if size[ra] < size[rb]:
            ra, rb = (rb, ra)
        parent[rb] = ra
        size[ra] += size[rb]
    nodes_by_val = {}
    for i, v in enumerate(vals):
        nodes_by_val.setdefault(v, []).append(i)
    sorted_vals = sorted(nodes_by_val.keys())
    answer = n
    for val in sorted_vals:
        nodes = nodes_by_val[val]
        for node in nodes:
            for nb in adj[node]:
                if vals[nb] <= val:
                    union(node, nb)
        comp_count = {}
        for node in nodes:
            root = find(node)
            comp_count[root] = comp_count.get(root, 0) + 1
        for cnt in comp_count.values():
            answer += cnt * (cnt - 1) // 2
    return answer

def solve(routes, source, target):
    if source == target:
        return 0
    stop_to_buses = defaultdict(list)
    for i, route in enumerate(routes):
        for stop in route:
            stop_to_buses[stop].append(i)
    if source not in stop_to_buses or target not in stop_to_buses:
        return -1
    visited_buses = set()
    visited_stops = set([source])
    q = deque()
    for bus in stop_to_buses[source]:
        q.append((bus, 1))
        visited_buses.add(bus)
    while q:
        bus, depth = q.popleft()
        for stop in routes[bus]:
            if stop == target:
                return depth
            if stop in visited_stops:
                continue
            visited_stops.add(stop)
            for nb_bus in stop_to_buses[stop]:
                if nb_bus not in visited_buses:
                    visited_buses.add(nb_bus)
                    q.append((nb_bus, depth + 1))
    return -1
