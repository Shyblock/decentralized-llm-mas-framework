def solve(words):
    if not words:
        return []
    n = len(words[0])
    prefix_map = {}
    for w in words:
        for i in range(n + 1):
            prefix = w[:i]
            prefix_map.setdefault(prefix, []).append(w)
    result = []

    def backtrack(step, path):
        if step == n:
            result.append(path[:])
            return
        prefix = ''.join((word[step] for word in path))
        for cand in prefix_map.get(prefix, []):
            path.append(cand)
            backtrack(step + 1, path)
            path.pop()
    for w in words:
        backtrack(1, [w])
    return result

def solve(routes, source, target):
    if source == target:
        return 0
    stop_to_routes = {}
    for i, route in enumerate(routes):
        for stop in route:
            stop_to_routes.setdefault(stop, set()).add(i)
    if source not in stop_to_routes or target not in stop_to_routes:
        return -1
    visited_routes = set()
    visited_stops = set([source])
    q = deque()
    for r in stop_to_routes[source]:
        visited_routes.add(r)
        q.append((r, 1))
    while q:
        route_idx, buses = q.popleft()
        if target in routes[route_idx]:
            return buses
        for stop in routes[route_idx]:
            if stop in visited_stops:
                continue
            visited_stops.add(stop)
            for nxt_route in stop_to_routes.get(stop, []):
                if nxt_route not in visited_routes:
                    visited_routes.add(nxt_route)
                    q.append((nxt_route, buses + 1))
    return -1

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        ratio = value * weight[y] / weight[x]
        parent[root_x] = root_y
        weight[root_x] = ratio
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])
    return result

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        ratio = weight[y] * value / weight[x]
        parent[root_x] = root_y
        weight[root_x] = ratio
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])
    return result

        for c in range(26):

    prev_idx = [[-1] * 26 for _ in range(n)]
    for i in range(n):
        if i > 0:
            for c in range(26):
                prev_idx[i][c] = prev_idx[i - 1][c]
        prev_idx[i][ord(s[i]) - 97] = i
    dp = [[0] * n for _ in range(n)]
    for i in range(n):
        dp[i][i] = 1
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j]:
                c = ord(s[i]) - 97
                i_next = next_idx[i + 1][c]
                j_prev = prev_idx[j - 1][c]
                if i_next > j_prev:
                    dp[i][j] = dp[i + 1][j - 1] * 2 + 2
                elif i_next == j_prev:
                    dp[i][j] = dp[i + 1][j - 1] * 2 + 1
                else:
                    dp[i][j] = dp[i + 1][j - 1] * 2 - dp[i_next + 1][j_prev - 1]
            else:
                dp[i][j] = dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1]

def solve(nums, k):
    total = sum(nums)
    if k <= 0 or total % k != 0:
        return False
    target = total // k
    nums.sort(reverse=True)
    if nums and nums[0] > target:
        return False
    n = len(nums)
    if k == 1:
        return True
    memo = {}

    def dfs(k_remaining, start, cur_sum, used):
        if k_remaining == 1:
            return True
        if cur_sum == target:
            res = dfs(k_remaining - 1, 0, 0, used)
            memo[used] = res
            return res
        if used in memo:
            return memo[used]
        for i in range(start, n):
            if not used >> i & 1 and cur_sum + nums[i] <= target:
                used2 = used | 1 << i
                if dfs(k_remaining, i + 1, cur_sum + nums[i], used2):
                    memo[used2] = True
                    return True
        memo[used] = False
        return False
    return dfs(k, 0, 0, 0)

def solve(equations, values, queries):
    graph = defaultdict(list)
    for (a, b), v in zip(equations, values):
        graph[a].append((b, v))
        graph[b].append((a, 1.0 / v))

    def bfs(src, dst):
        if src == dst:
            return 1.0
        visited = set([src])
        q = deque([(src, 1.0)])
        while q:
            node, prod = q.popleft()
            for neigh, w in graph[node]:
                if neigh in visited:
                    continue
                if neigh == dst:
                    return prod * w
                visited.add(neigh)
                q.append((neigh, prod * w))
        return -1.0
    results = []
    for a, b in queries:
        if a not in graph or b not in graph:
            results.append(-1.0)
        else:
            results.append(bfs(a, b))
    return results

    i = 1
    while i < n - 1:
        if arr[i - 1] < arr[i] > arr[i + 1]:
            left = i - 1
            while left > 0 and arr[left - 1] < arr[left]:
                left -= 1
            right = i + 1
            while right + 1 < n and arr[right] > arr[right + 1]:
                right += 1
            cur_len = right - left + 1
