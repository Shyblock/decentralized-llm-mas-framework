            prefix_map[w[:i]].append(w)
    results = []

    def backtrack(step, square):
        if step == n:
            results.append(square[:])
            return
        prefix = ''.join((square[i][step] for i in range(step)))
        for candidate in prefix_map.get(prefix, []):
            square.append(candidate)
            backtrack(step + 1, square)
            square.pop()
    for w in words:
        backtrack(1, [w])
    return results

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        parent[root_x] = root_y
        weight[root_x] = value * weight[y] / weight[x]
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])
    return result

def solve(jobs, k):
    if not jobs:
        return 0
    n = len(jobs)
    if k >= n:
        return max(jobs)
    jobs_sorted = sorted(jobs, reverse=True)
    low = max(jobs_sorted)
    high = sum(jobs_sorted)

    def can(limit):
        loads = [0] * k

        def dfs(idx):
            if idx == n:
                return True
            cur = jobs_sorted[idx]
            seen = set()
            for i in range(k):
                if loads[i] + cur <= limit and loads[i] not in seen:
                    seen.add(loads[i])
                    loads[i] += cur
                    if dfs(idx + 1):
                        return True
                    loads[i] -= cur
                if loads[i] == 0:
                    break
            return False
        return dfs(0)
    while low < high:
        mid = (low + high) // 2
        if can(mid):
            high = mid
        else:
            low = mid + 1
    return low

    dist[src] = 0
    cnt[src] = 1
    heap = [(0, src)]
    while heap:
        d, u = heapq.heappop(heap)
        if d != dist[u]:
            continue
        for v, w in adj[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                cnt[v] = cnt[u]
                heapq.heappush(heap, (nd, v))
            elif nd == dist[v]:
                cnt[v] = (cnt[v] + cnt[u]) % MOD
    return cnt

def solve(equations):
    parent = {}
    rank = {}

    def find(x):
        if parent[x] != x:
            parent[x] = find(parent[x])
        return parent[x]

    def union(x, y):
        rx, ry = (find(x), find(y))
        if rx == ry:
            return
        if rank[rx] < rank[ry]:
            parent[rx] = ry
        elif rank[rx] > rank[ry]:
            parent[ry] = rx
        else:
            parent[ry] = rx
            rank[rx] += 1
    for eq in equations:
        var1 = eq[0]
        var2 = eq[3]
        if var1 not in parent:
            parent[var1] = var1
            rank[var1] = 0
        if var2 not in parent:
            parent[var2] = var2
            rank[var2] = 0

        if eq[1:3] == '==':
            union(eq[0], eq[3])
    for eq in equations:
        if eq[1:3] == '!=':
            if find(eq[0]) == find(eq[3]):
                return False
    return True

def solve(n, k):

    def backtrack(start, path, res):
        if len(path) == k:
            res.append(path[:])
            return
        for i in range(start, n + 1):
            path.append(i)
            backtrack(i + 1, path, res)
            path.pop()
    if k == 0:
        return [[]]
    if k > n:
        return []
    result = []
    backtrack(1, [], result)
    return result

def solve(vals, edges):
    n = len(vals)
    adj = [[] for _ in range(n)]
    for u, v in edges:
        adj[u].append(v)
        adj[v].append(u)

    class DSU:
        __slots__ = ('parent', 'rank')

        def __init__(self, size):
            self.parent = list(range(size))
            self.rank = [0] * size

        def find(self, x):
            while self.parent[x] != x:
                self.parent[x] = self.parent[self.parent[x]]
                x = self.parent[x]
            return x

        def union(self, a, b):
            ra, rb = (self.find(a), self.find(b))
            if ra == rb:
                return
            if self.rank[ra] < self.rank[rb]:
                ra, rb = (rb, ra)
            self.parent[rb] = ra
            if self.rank[ra] == self.rank[rb]:
                self.rank[ra] += 1
    order = sorted(range(n), key=lambda i: vals[i])
    dsu = DSU(n)
    active = [False] * n
    ans = n
    i = 0
    while i < n:
        cur_val = vals[order[i]]
        same = []
        while i < n and vals[order[i]] == cur_val:
            node = order[i]
            active[node] = True
            same.append(node)
            i += 1
        for node in same:
            for nb in adj[node]:
                if active[nb]:
                    dsu.union(node, nb)
        comp_cnt = {}
        for node in same:
            root = dsu.find(node)
            comp_cnt[root] = comp_cnt.get(root, 0) + 1
        for c in comp_cnt.values():
            ans += c * (c - 1) // 2
