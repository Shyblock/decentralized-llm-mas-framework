        return []
    n = len(words[0])
    prefix_dict = {}
    for w in words:
        for i in range(n + 1):
            prefix = w[:i]
            prefix_dict.setdefault(prefix, []).append(w)
    result = []

    def backtrack(step, square):
        if step == n:
            result.append(square[:])
            return
        prefix = ''.join((square[i][step] for i in range(step)))
        for cand in prefix_dict.get(prefix, []):
            square.append(cand)
            backtrack(step + 1, square)
            square.pop()
    for w in words:
        backtrack(1, [w])
    return result

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        ratio = weight[y] * value / weight[x]
        parent[root_x] = root_y
        weight[root_x] = ratio
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])
    return result

    visited[0][0] = True
    while heap:
        t, x, y = heapq.heappop(heap)
        if x == n - 1 and y == n - 1:
            return t
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            nx, ny = (x + dx, y + dy)
            if 0 <= nx < n and 0 <= ny < n and (not visited[nx][ny]):

    for ch in s:
        if ch == '(':

        elif ch == ')':
            if left_rem > 0:
                left_rem -= 1
            else:
                right_rem += 1
    res = set()
    visited = set()
    n = len(s)

    def dfs(i, lrem, rrem, bal, path):
        if i == n:
            if lrem == 0 and rrem == 0 and (bal == 0):
                res.add(''.join(path))

        if key in visited:
            return

            if lrem > 0:
                dfs(i + 1, lrem - 1, rrem, bal, path)
            path.append(ch)
            dfs(i + 1, lrem, rrem, bal + 1, path)

        elif ch == ')':

                path.pop()
        else:
            path.append(ch)
            dfs(i + 1, lrem, rrem, bal, path)
            path.pop()
    dfs(0, left_rem, right_rem, 0, [])

    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        ratio = value * weight[y] / weight[x]
        parent[root_x] = root_y
        weight[root_x] = ratio
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])

        add(x)

    if n == 0:
        return 0
    dp = [[0] * n for _ in range(n)]
    for i in range(n):
        dp[i][i] = 1
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j]:
                low = i + 1
                high = j - 1
                while low <= high and s[low] != s[i]:
                    low += 1
                while low <= high and s[high] != s[i]:
                    high -= 1
                if low > high:
                    dp[i][j] = dp[i + 1][j - 1] * 2 + 2
                elif low == high:
                    dp[i][j] = dp[i + 1][j - 1] * 2 + 1
                else:
                    dp[i][j] = dp[i + 1][j - 1] * 2 - dp[low + 1][high - 1]
            else:
                dp[i][j] = dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1]
    return dp[0][n - 1]

def solve(matrix, target):

    def binary_search(row, target):
        lo, hi = (0, len(row) - 1)
        while lo <= hi:
            mid = (lo + hi) // 2
            val = row[mid]
            if val == target:
                return True
            elif val < target:
                lo = mid + 1
            else:
                hi = mid - 1
        return False
    for row in matrix:
        if not row:
            continue
        if row[0] <= target <= row[-1]:
            if binary_search(row, target):
                return True
    return False

    count = 0

    if source == target:
        return 0
    stop_to_buses = defaultdict(list)
    for i, stops in enumerate(routes):
        for s in stops:
            stop_to_buses[s].append(i)
    if source not in stop_to_buses or target not in stop_to_buses:
        return -1
    visited_buses = set()
    q = deque()
    for bus in stop_to_buses[source]:
        visited_buses.add(bus)
        q.append((bus, 1))
    while q:
        bus, dist = q.popleft()
        if target in routes[bus]:
            return dist
        for stop in routes[bus]:
            for nb in stop_to_buses[stop]:
                if nb not in visited_buses:
                    visited_buses.add(nb)
                    q.append((nb, dist + 1))
    return -1

def solve(equations, values, queries):
    graph = defaultdict(list)
    for (a, b), v in zip(equations, values):
        graph[a].append((b, v))
        graph[b].append((a, 1.0 / v))

    def bfs(src, dst):
        if src not in graph or dst not in graph:
            return -1.0
        if src == dst:
            return 1.0
        visited = set([src])
        q = deque([(src, 1.0)])
        while q:
            node, prod = q.popleft()
            for nei, w in graph[node]:
                if nei in visited:
                    continue
                if nei == dst:
                    return prod * w
                visited.add(nei)
                q.append((nei, prod * w))
        return -1.0
    return [bfs(a, b) for a, b in queries]

        return 0

        if arr[i] > arr[i - 1]:
            left[i] = left[i - 1] + 1
    right = [0] * n
    for i in range(n - 2, -1, -1):
        if arr[i] > arr[i + 1]:
            right[i] = right[i + 1] + 1
    max_len = 0
    for i in range(n):
        if left[i] > 0 and right[i] > 0:
            cur = left[i] + right[i] + 1
