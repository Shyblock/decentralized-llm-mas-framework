    prefix_map = {}
    for w in words:
        for i in range(n + 1):
            pref = w[:i]
            prefix_map.setdefault(pref, []).append(w)
    results = []

    def backtrack(path):
        if len(path) == n:
            results.append(path[:])
            return
        idx = len(path)
        prefix = ''.join((word[idx] for word in path))
        for cand in prefix_map.get(prefix, []):
            path.append(cand)
            backtrack(path)
            path.pop()
    for w in words:
        backtrack([w])
    return results

    if n == 0:
        return 0
    prefix = [0] * (n + 1)
    for i in range(n):
        prefix[i + 1] = prefix[i] + nums[i]
    left = [-1] * n
    right = [n] * n
    stack = []
    for i in range(n):
        while stack and nums[stack[-1]] >= nums[i]:
            stack.pop()
        left[i] = stack[-1] if stack else -1
        stack.append(i)
    stack.clear()
    for i in range(n - 1, -1, -1):
        while stack and nums[stack[-1]] > nums[i]:
            stack.pop()
        right[i] = stack[-1] if stack else n
        stack.append(i)
    max_product = 0
    for i in range(n):
        total = prefix[right[i]] - prefix[left[i] + 1]
        product = nums[i] * total
        if product > max_product:
            max_product = product
    return max_product

        return []

    dirs = [(1, 0), (-1, 0), (0, 1), (0, -1)]

    def bfs(starts):
        reachable = [[False] * n for _ in range(m)]
        stack = list(starts)
        for i, j in starts:
            reachable[i][j] = True
        while stack:
            i, j = stack.pop()
            cur_h = heights[i][j]
            for di, dj in dirs:
                ni, nj = (i + di, j + dj)
                if 0 <= ni < m and 0 <= nj < n and (not reachable[ni][nj]) and (heights[ni][nj] >= cur_h):
                    reachable[ni][nj] = True
                    stack.append((ni, nj))
        return reachable
    top_left_starts = [(i, 0) for i in range(m)] + [(0, j) for j in range(1, n)]
    bottom_right_starts = [(i, n - 1) for i in range(m)] + [(m - 1, j) for j in range(n - 1)]
    reach_tl = bfs(top_left_starts)
    reach_br = bfs(bottom_right_starts)
    result = []
    for i in range(m):
        for j in range(n):
            if reach_tl[i][j] and reach_br[i][j]:
                result.append([i, j])

        return 0

        for s in stops:
            stop_to_buses[s].append(i)
    if source not in stop_to_buses or target not in stop_to_buses:
        return -1
    visited_buses = set()
    q = deque()
    for bus in stop_to_buses[source]:
        visited_buses.add(bus)
        q.append((bus, 1))
    while q:
        bus, depth = q.popleft()
        if target in routes[bus]:
            return depth
        for stop in routes[bus]:
            for nb in stop_to_buses[stop]:
                if nb not in visited_buses:
                    visited_buses.add(nb)
                    q.append((nb, depth + 1))
    return -1

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        parent[root_x] = root_y
        weight[root_x] = value * weight[y] / weight[x]
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    result = []
    for a, b in queries:
        if a not in parent or b not in parent:
            result.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            result.append(-1.0)
        else:
            result.append(weight[a] / weight[b])
    return result

            return True

            if prev == nums[i]:

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root, w = find(orig_parent)
            parent[x] = root
            weight[x] *= w
        return (parent[x], weight[x])

    def union(x, y, value):
        add(x)
        add(y)
        root_x, w_x = find(x)
        root_y, w_y = find(y)
        if root_x == root_y:
            return
        parent[root_x] = root_y
        weight[root_x] = value * w_y / w_x
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    res = []
    for a, b in queries:
        if a not in parent or b not in parent:
            res.append(-1.0)
            continue
        root_a, w_a = find(a)
        root_b, w_b = find(b)
        if root_a != root_b:
            res.append(-1.0)
        else:
            res.append(w_a / w_b)
    return res
