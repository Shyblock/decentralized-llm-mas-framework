        for i in range(n + 1):
            prefix = w[:i]
            prefix_map.setdefault(prefix, []).append(w)
    results = []

    def backtrack(step, square):
        if step == n:
            results.append(square[:])
            return
        prefix = ''.join((square[i][step] for i in range(step)))
        for cand in prefix_map.get(prefix, []):
            square.append(cand)
            backtrack(step + 1, square)
            square.pop()
    for w in words:
        backtrack(1, [w])
    return results

def solve(equations, values, queries):

    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

        if parent[x] != x:
            orig_parent = parent[x]
            root = find(orig_parent)
            weight[x] *= weight[orig_parent]
            parent[x] = root
        return parent[x]

    def union(x, y, value):
        add(x)
        add(y)
        root_x = find(x)
        root_y = find(y)
        if root_x == root_y:
            return
        ratio = value * weight[y] / weight[x]
        parent[root_x] = root_y
        weight[root_x] = ratio
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    res = []
    for a, b in queries:
        if a not in parent or b not in parent:
            res.append(-1.0)
            continue
        root_a = find(a)
        root_b = find(b)
        if root_a != root_b:
            res.append(-1.0)
        else:

                dp[i][j] = dp[i + 1][j] + dp[i][j - 1] - dp[i + 1][j - 1]
            else:
                low = i + 1
                high = j - 1

                if low > high:

                else:

def solve(nums, k):
    total = sum(nums)
    if k <= 0 or total % k != 0:
        return False
    target = total // k

        return True

        return False

    def backtrack(start, cur_sum, buckets_done):
        if buckets_done == k - 1:
            return True
        if cur_sum == target:
            return backtrack(0, 0, buckets_done + 1)
        prev = -1
        i = start
        while i < n:
            if not used[i] and cur_sum + nums[i] <= target and (nums[i] != prev):
                used[i] = True
                if backtrack(i + 1, cur_sum + nums[i], buckets_done):
                    return True
                used[i] = False

                    return False
                if cur_sum + nums[i] == target:
                    return False
                prev = nums[i]
            i += 1
        return False
    return backtrack(0, 0, 0)

def solve(equations, values, queries):
    parent = {}
    weight = {}

    def add(x):
        if x not in parent:
            parent[x] = x
            weight[x] = 1.0

    def find(x):
        if parent[x] != x:
            orig_parent = parent[x]
            root, w = find(orig_parent)
            parent[x] = root
            weight[x] *= w
        return (parent[x], weight[x])

    def union(x, y, value):
        add(x)
        add(y)
        root_x, w_x = find(x)
        root_y, w_y = find(y)
        if root_x == root_y:
            return
        parent[root_x] = root_y
        weight[root_x] = w_y * value / w_x
    for (a, b), v in zip(equations, values):
        union(a, b, v)
    res = []
    for a, b in queries:
        if a not in parent or b not in parent:
            res.append(-1.0)
            continue
        root_a, w_a = find(a)
        root_b, w_b = find(b)
        if root_a != root_b:
            res.append(-1.0)
        else:
            res.append(w_a / w_b)
    return res
